test('pin a converged mission clears the pin and drives nothing', async () => {
  addWatchedProject(project);
  setConductorEnabled(project, true);
  await forgeApprovedActive(); // an otherwise-actionable mission that must NOT be fallen back to
  const converged = await forgeMission(project, { session: 's1', title: 'already converged', criteria: ['c'] });
  setCriterionMet(project, listCriteria(project, converged.missionId)[0].id, true);
  expect(getMission(project, converged.missionId)?.status).toBe('converged');

  setConductorTargetMission(project, converged.missionId);
  const r = await runConductorPass(project, { invoke: okInvoke });
  expect(r.ran).toBe(false);
  expect(r.reason).toBe('target-cleared');
  expect(invokeCalls).toBe(0);
  expect(getConductorTargetMission(project)).toBe(null);
});

test('pin an abandoned mission clears the pin and drives nothing', async () => {

  setConductorEnabled(project, true);
  await forgeApprovedActive();
