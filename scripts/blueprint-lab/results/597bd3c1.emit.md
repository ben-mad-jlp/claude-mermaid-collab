## Blueprint: Throttle + orchestrator wiring for the landed-epic sweep

### Problem

`src/services/landed-epic-sweep.ts` exports `reconcileLandedEpics` (stamps/completes `[LAND]` leaves for converged-mission epics that have fully landed) and `gcEpicBranches` (deletes `collab/epic/<id8>` branches proven `ahead===0`, flagging the rest). Both are fully tested in isolation (`src/services/__tests__/landed-epic-sweep.test.ts`) but **neither is throttled nor wired into the orchestrator tick** — grep confirms zero callers outside the test file. This leaf follows the exact pattern established by `ARCHIVAL_SWEEP_INTERVAL_MS` / `shouldRunArchivalSweep` / `runArchivalSweep` (`src/services/archival-sweep.ts:16-64`) and its wiring in `orchestrator-live.ts:38,306-311,379-380,571-580`.

### Change 1 — `src/services/landed-epic-sweep.ts`: add throttle + combined pass wrapper

Append after `gcEpicBranches` (after line 154):

```ts
/** Minimum spacing between PERIODIC landed-epic sweep passes for a single project.
 *  Hygiene (land-leaf finalization + branch GC), not claim/build-latency-sensitive —
 *  same throttle shape as ARCHIVAL_SWEEP_INTERVAL_MS (archival-sweep.ts:20). */
export const LANDED_EPIC_SWEEP_INTERVAL_MS = 300_000; // 5 min

const lastLandedEpicSweepMs = new Map<string, number>();

export function shouldRunLandedEpicSweep(project: string, now: number = Date.now()): boolean {
  const last = lastLandedEpicSweepMs.get(project);
  if (last !== undefined && now - last < LANDED_EPIC_SWEEP_INTERVAL_MS) return false;
  lastLandedEpicSweepMs.set(project, now);
  return true;
}

export function _resetLandedEpicSweepThrottle(project?: string): void {
  if (project === undefined) lastLandedEpicSweepMs.clear();
  else lastLandedEpicSweepMs.delete(project);
}

export interface LandedEpicSweepPassResult {
  reconcile: LandedEpicSweepResult;
  gc: GcEpicBranchesResult;
}

/** Combined periodic pass: reconcile land-leaves, then GC fully-landed branches.
 *  Self-gated like runArchivalSweep (archival-sweep.ts:47-64) — `force` lets tests/
 *  direct callers bypass the throttle window; the orchestrator tick wiring below
 *  always passes force:true since it already consulted shouldRunLandedEpicSweep as
 *  the outer gate (same two-seam shape as archival/reconcile). */
export async function runLandedEpicSweepPass(
  project: string,
  opts: { probe?: GitProbe; runner?: BranchGcRunner; baseRef?: string; now?: number; force?: boolean } = {},
): Promise<LandedEpicSweepPassResult> {
  const now = opts.now ?? Date.now();
  if (!opts.force && !shouldRunLandedEpicSweep(project, now)) {
    return { reconcile: { reconciled: [], skipped: 0 }, gc: { deleted: [], flagged: [], skipped: 0 } };
  }
  const reconcile = await reconcileLandedEpics(project, { probe: opts.probe, baseRef: opts.baseRef });
  const gc = gcEpicBranches(project, { probe: opts.probe, runner: opts.runner, baseRef: opts.baseRef });
  return { reconcile, gc };
}
```

`GitProbe` and `BranchGcRunner` are already in scope (imported/declared above in this file). No new imports needed.
