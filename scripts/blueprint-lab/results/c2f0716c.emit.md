### Part 1 — MCP tool

Create a new domain module **`src/mcp/leaf-blueprint-tools.ts`**, following the `WORKGRAPH_TOOL_DEFS`/`handleWorkgraphTool` pattern in `src/mcp/workgraph-tools.ts:180-306` exactly (handler-less defs array + a `handle*Tool(name, args): Promise<string|null>` dispatcher returning `null` for unrecognized names).

```ts
import { getLeafBlueprint, editLeafRequirement } from '../services/worker-ledger.js';
import { validateRequirementShape } from '../services/diff-contract.js'; // see note below — export this
import { getWebSocketHandler } from '../services/ws-handler-manager.js';

export const LEAF_BLUEPRINT_TOOL_DEFS = [
  {
    name: 'edit_leaf_requirement',
    description:
      "Replace ONE entry in a leaf's editable DiffContract.requirements[] at `index` (e.g. to fix a wrong file/symbol citation without re-blueprinting). Seeds the editable overlay from the base blueprint if this is the leaf's first edit. Never touches the append-only ledger — takes effect on the leaf's NEXT claim (reattach-blueprint resume), not an in-flight run. Does NOT reset poison-loop detection (isPoisonLooped hashes the base ledger blueprint text, unaffected by this edit).",
    inputSchema: {
      type: 'object',
      properties: {
        project: { type: 'string' },
        leafId: { type: 'string' },
        index: { type: 'number', description: '0-based index into the CURRENT requirements[] to replace.' },
        replacement: {
          type: 'object',
          description: 'A full DiffRequirement object: {kind:"symbol-present",file,symbol,description} | {kind:"named-test",testFile,testName,mechanical:true} | {kind:"threshold",source,metric,comparison,value,mechanical:true}',
        },
      },
      required: ['project', 'leafId', 'index', 'replacement'],
    },
  },
];

export async function handleLeafBlueprintTool(name: string, args: any): Promise<string | null> {
  switch (name) {
    case 'edit_leaf_requirement': {
      const { project, leafId, index, replacement } = args as {
        project: string; leafId: string; index: number; replacement: unknown;
      };
      if (!project || !leafId || typeof index !== 'number' || !replacement) {
        throw new Error('Missing required: project, leafId, index, replacement');
      }
      const validated = validateRequirementShape(replacement);
      if (!validated) throw new Error('edit_leaf_requirement: replacement is not a valid DiffRequirement');
      const ok = editLeafRequirement(leafId, index, validated);
      if (!ok) throw new Error('edit_leaf_requirement: no editable blueprint for this leaf, or index out of range');
      const row = getLeafBlueprint(leafId);
      getWebSocketHandler()?.broadcast({ type: 'session_todos_updated', project, session: '' });
      return JSON.stringify({ leafId, specRev: row?.specRev ?? null, specJson: row?.specJson ?? null }, null, 2);
    }
    default:
      return null;
  }
}
```

### Part 2 — end-to-end poison-trap test

The trap this test proves: **editing a requirement citation must NOT let a leaf silently escape B5's poison-loop detection.** `isPoisonLooped` (`src/services/reserve-leaf.ts:109-135`) hashes each run's blueprint via the **base ledger `outputText`** (`rowsToRuns`, `reserve-leaf.ts:84-89`), never `leaf_blueprint.specJson`. Since `leaf-executor.ts:2531` only calls `persistBlueprintBase` (which seeds `specJson`) `if (!reattach && !inRunCarry)`, and a reattach never writes a fresh `worker_ledger` blueprint node (`leaf-executor.ts:2479-2483`, synthetic `NodeResult` with `durationMs:0`), a citation-only edit changes nothing the poison detector can see. Two consecutive same-blueprint rejections must still register as poisoned even after an `edit_leaf_requirement` call in between.

New file **`src/services/__tests__/edit-leaf-requirement.test.ts`**, mirroring `src/services/__tests__/reserve-leaf.test.ts:1-50` exactly (same `bun:test`/`bun:sqlite` harness, `_closeProject`/`_closeLedgerDb` teardown, `recordRejectedRun` helper copied verbatim):

```ts
import { describe, test, expect, beforeEach, afterEach } from 'bun:test';
import { mkdtempSync, rmSync } from 'node:fs';
import { tmpdir } from 'node:os';
import { join } from 'node:path';
import { createTodo, _closeProject } from '../todo-store';
import { recordNode, _closeLedgerDb, recordLeafBlueprint, editLeafRequirement, getLeafBlueprint } from '../worker-ledger';
import { isPoisonLooped } from '../reserve-leaf';
import { renderContract, type DiffContract } from '../diff-contract';

let project: string;
beforeEach(() => { project = mkdtempSync(join(tmpdir(), 'edit-leaf-req-')); });
afterEach(() => { _closeProject(project); _closeLedgerDb(); rmSync(project, { recursive: true, force: true }); });

async function makeLeaf(): Promise<string> { /* same as reserve-leaf.test.ts:31-39 */ }
function recordRejectedRun(leafId: string, blueprint: string, runIndex: number): void { /* copy reserve-leaf.test.ts:44-50 */ }

const BASE_CONTRACT: DiffContract = {
  schemaVersion: 2, estimatedFiles: 1, estimatedTasks: 1, nonEnumerableFanout: false,
  filesToCreate: [], filesToEdit: ['a.ts'], tasks: [{ id: 't1', files: ['a.ts'], description: 'x' }],
  leafKind: 'fix',
  requirements: [{ kind: 'symbol-present', file: 'a.ts', symbol: 'wrongSymbol', description: 'stub' }],
  outOfScope: [],
};

describe('edit_leaf_requirement poison-trap', () => {
  test('editing a requirement citation does NOT reset poison-loop detection', async () => {
    const leaf = await makeLeaf();
    const blueprintText = `plan\n\n${renderContract(BASE_CONTRACT)}`;

    // Two rejections with the SAME base blueprint → poisoned.
    recordRejectedRun(leaf, blueprintText, 0);
    recordRejectedRun(leaf, blueprintText, 1);
    expect(isPoisonLooped(project, leaf)).toBe(true);

    // A human/conductor now calls edit_leaf_requirement to fix the wrong symbol citation.
    recordLeafBlueprint({ leafId: leaf, project });
    const edited = editLeafRequirement(leaf, 0, { kind: 'symbol-present', file: 'a.ts', symbol: 'rightSymbol', description: 'fixed' });
    expect(edited).toBe(true);
    const row = getLeafBlueprint(leaf);
    expect(row?.specRev).toBe(1);
    expect(row?.specJson).toContain('rightSymbol');

    // The edit does not touch worker_ledger.outputText — poison state is UNCHANGED.
    expect(isPoisonLooped(project, leaf)).toBe(true);

    // A THIRD rejection with the SAME (unedited) base blueprint is still the same lineage —
    // still poisoned, proving the edit was not a silent escape hatch.
    recordRejectedRun(leaf, blueprintText, 2);
    expect(isPoisonLooped(project, leaf)).toBe(true);
  });
});
```
