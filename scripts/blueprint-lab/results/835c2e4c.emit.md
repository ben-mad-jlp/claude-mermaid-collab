export function recordLeafBlueprint(
  e: { leafId: string; project: string; epicBaseSha?: string | null; specJson?: string | null },
  now: number = Date.now(),
): void {
  try {
    openDb().prepare(
      `INSERT INTO leaf_blueprint (leafId, project, epicBaseSha, specJson, originalSpecJson, specRev, recordedAt)

### 4. New accessor: `editLeafBlueprintSpec` — updates `specJson`, increments `specRev`

Add directly below `getLeafBlueprint` (after `worker-ledger.ts:536`):

```ts
/** Edit a leaf's durable blueprint spec — overwrites `specJson` and increments `specRev`.
 *  `originalSpecJson` is left untouched so {@link restoreLeafBlueprintSpec} can always
 *  revert. No-op (returns false) if no row exists for this leaf. Best-effort. */
export function editLeafBlueprintSpec(leafId: string, specJson: string, now: number = Date.now()): boolean {
  try {
    const res = openDb().prepare(
      `UPDATE leaf_blueprint SET specJson=?, specRev=specRev+1, recordedAt=? WHERE leafId=?`,
    ).run(specJson, now, leafId);
    return res.changes > 0;
  } catch { return false; }
}
```

### 5. New accessor: `restoreLeafBlueprintSpec` — reverts `specJson` to `originalSpecJson`

```ts
/** Restore a leaf's blueprint spec to its originally-recorded value, discarding any edits.
 *  Resets `specRev` to 0. No-op (returns false) if no row exists, or if `originalSpecJson`
 *  was never set (nothing to restore to). Best-effort. */
export function restoreLeafBlueprintSpec(leafId: string, now: number = Date.now()): boolean {
  try {
    const res = openDb().prepare(
      `UPDATE leaf_blueprint SET specJson=originalSpecJson, specRev=0, recordedAt=?
       WHERE leafId=? AND originalSpecJson IS NOT NULL`,
    ).run(now, leafId);
    return res.changes > 0;
  } catch { return false; }
}
```
